fn main() -> Result<(), String> {
    let arguments: Vec<_> = std::env::args().collect();
    let destination = std::path::Path::new(arguments.get(1).ok_or("Missing directory")?);
    std::fs::create_dir_all(destination).map_err(|error| error.to_string())?;
    let source = std::fs::read_to_string(arguments.get(2).ok_or("Missing source")?).map_err(|error| error.to_string())?;
    let request = pseudolearn_api::ExecuteRequest { program: source.clone(), ..Default::default() };
    let request = pseudolearn_api::conformance_build_request(&request);
    let original = pseudolearn_api::build_benchmark_object(&request).map_err(|error| format!("{error:?}"))?;
    let settings = pseudolearn_aot_rt::decode_metadata(&original.metadata).ok_or("Invalid metadata")?.settings;
    let module = pseudolearn_api::compile_source_to_module(&source, &request.profile, &request.strictness).map_err(|error| format!("{error:?}"))?;
    for (name, policy) in [("all-live", pseudolearn_codegen::ParameterRoots::AllLive), ("borrowed", pseudolearn_codegen::ParameterRoots::Borrowed)] {
        let target = pseudolearn_codegen::TargetConfig::host().with_inspection().with_parameter_roots(policy);
        let compiled = pseudolearn_codegen::compile_module_to_benchmark_object(&module, &target, &settings).map_err(|error| error.to_string())?;
        let entry = compiled.function_symbols.iter().find(|symbol| symbol.name.trim_start_matches('@') == "main").ok_or("Missing entry")?.symbol.clone();
        std::fs::write(destination.join(format!("{name}.o")), &compiled.bytes).map_err(|error| error.to_string())?;
        std::fs::write(destination.join(format!("{name}.clif")), compiled.inspection.join("\n")).map_err(|error| error.to_string())?;
        let metadata = pseudolearn_aot_rt::ExecutableMetadata { settings: settings.clone(), sites: compiled.fault_sites, classes: compiled.classes };
        let configuration = serde_json::json!({"workload":"permute","source":source,"size":6,"input":"6\n","expected":"8660\n","metadata":pseudolearn_aot_rt::encode_metadata(&metadata),"process":0,"iterations":30});
        std::fs::write(destination.join(format!("{name}.json")), serde_json::to_vec_pretty(&configuration).map_err(|error| error.to_string())?).map_err(|error| error.to_string())?;
        let template = std::fs::read_to_string(arguments.get(3).ok_or("Missing launcher")?).map_err(|error| error.to_string())?;
        std::fs::write(destination.join(format!("{name}.c")), template.replace("@@ENTRY_SYMBOL@@", &entry)).map_err(|error| error.to_string())?;
    }
    Ok(())
}
