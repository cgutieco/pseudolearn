fn main() -> Result<(), String> {
    let arguments: Vec<String> = std::env::args().collect();
    let text = std::fs::read_to_string(&arguments[1]).map_err(|error| error.to_string())?;
    let request = pseudolearn_api::ExecuteRequest { program: text, ..Default::default() };
    let build = pseudolearn_api::conformance_build_request(&request);
    let module = pseudolearn_api::compile_source_to_module(&build.program, &build.profile, &build.strictness)
        .map_err(|errors| format!("{errors:?}"))?;
    let bytes = std::fs::read(&arguments[2]).map_err(|error| error.to_string())?;
    let previous = pseudolearn_aot_rt::decode_metadata(&bytes).ok_or("metadata")?;
    let target = if std::env::var("PL_INSPECT").is_ok() { pseudolearn_codegen::TargetConfig::host().with_inspection() } else { pseudolearn_codegen::TargetConfig::host() };
    let mut timings = Vec::new();
    let mut compiled = None;
    for _ in 0..15 {
        let started = std::time::Instant::now();
        let object = pseudolearn_codegen::compile_module_to_benchmark_object(&module, &target, &previous.settings)
            .map_err(|error| error.to_string())?;
        timings.push(started.elapsed().as_secs_f64() * 1000.0);
        compiled = Some(object);
    }
    let compiled = compiled.ok_or("none")?;
    if std::env::var("PL_TIMING").is_ok() { let _ = cranelift_codegen::timing::take_current(); let started = std::time::Instant::now(); let _again = pseudolearn_codegen::compile_module_to_benchmark_object(&module, &target, &previous.settings).map_err(|error| error.to_string())?; eprintln!("wall {:?}", started.elapsed()); eprintln!("{}", cranelift_codegen::timing::take_current()); }
    timings.sort_by(f64::total_cmp);
    if let Ok(path) = std::env::var("PL_INSPECT") { std::fs::write(path, compiled.inspection.join("\n")).map_err(|error| error.to_string())?; }
    std::fs::write(&arguments[3], &compiled.bytes).map_err(|error| error.to_string())?;
    let metadata = pseudolearn_aot_rt::ExecutableMetadata {
        settings: previous.settings,
        sites: compiled.fault_sites,
        classes: compiled.classes,
    };
    std::fs::write(&arguments[4], pseudolearn_aot_rt::encode_metadata(&metadata)).map_err(|error| error.to_string())?;
    let entry = compiled.function_symbols.iter().find(|symbol| symbol.name.trim_start_matches('@') == "main").map(|symbol| symbol.symbol.clone()).ok_or("main")?;
    println!("{entry} {:.3}", timings[timings.len() / 2]);
    Ok(())
}

use ceiling_kernels::variant::Experiment;
use pseudolearn_aot_rt::{AotConfig, AotSession, AotStreams, EntryFunction, decode_metadata};
use std::cell::RefCell;
use std::io::{Cursor, Write};
use std::rc::Rc;
use std::time::Instant;

#[derive(Clone, Default)]
struct Output(Rc<RefCell<Vec<u8>>>);

impl Write for Output {
    fn write(&mut self, bytes: &[u8]) -> std::io::Result<usize> {
        self.0.borrow_mut().extend_from_slice(bytes);
        Ok(bytes.len())
    }
    fn flush(&mut self) -> std::io::Result<()> {
        Ok(())
    }
}

struct Configuration {
    workload: String,
    size: i64,
    input: String,
    expected: String,
    metadata: Vec<u8>,
    process: u32,
    iterations: u32,
}

fn parse(bytes: &[u8]) -> Option<Configuration> {
    let value: serde_json::Value = serde_json::from_slice(bytes).ok()?;
    Some(Configuration {
        workload: value.get("workload")?.as_str()?.to_owned(),
        size: value.get("size")?.as_i64()?,
        input: value.get("input")?.as_str()?.to_owned(),
        expected: value.get("expected")?.as_str()?.to_owned(),
        metadata: value.get("metadata")?.as_array()?.iter().map(|byte| byte.as_u64().map(|b| b as u8)).collect::<Option<Vec<u8>>>()?,
        process: value.get("process")?.as_u64()? as u32,
        iterations: value.get("iterations")?.as_u64()? as u32,
    })
}

fn native(entry: EntryFunction, configuration: &Configuration) -> Result<u64, String> {
    let metadata = decode_metadata(&configuration.metadata).ok_or("Invalid metadata")?;
    let config = AotConfig { limits: metadata.settings.limits, seed: metadata.settings.seed, boolean_words: metadata.settings.boolean_words };
    let output = Output::default();
    let mut session = AotSession::new(config, metadata.sites, AotStreams {
        input: Box::new(Cursor::new(configuration.input.clone().into_bytes())),
        output: Box::new(output.clone()),
    });
    for (index, class) in metadata.classes.iter().enumerate() {
        session.engine_mut().register_class(index as u32, class).map_err(|error| format!("{error:?}"))?;
    }
    if metadata.settings.forced_collection {
        session.engine_mut().force_collection();
    }
    let started = Instant::now();
    entry(session.context_pointer());
    let elapsed = started.elapsed();
    if !pseudolearn_aot_rt::settle_session(&mut session) {
        return Err("Native allocation state could not be settled".to_owned());
    }
    let text = String::from_utf8(output.0.borrow().clone()).map_err(|error| error.to_string())?;
    if session.context().fault_status != 0 || text != configuration.expected {
        return Err("Native result or termination mismatch".to_owned());
    }
    Ok(elapsed.as_nanos() as u64)
}

fn ceiling(configuration: &Configuration) -> Result<u64, String> {
    let mut experiment = Experiment::new(4).map_err(|error| format!("{error:?}"))?;
    let started = Instant::now();
    let result = experiment.run(&configuration.workload, std::hint::black_box(configuration.size)).map_err(|error| format!("{error:?}"))?;
    let elapsed = started.elapsed();
    if result != configuration.expected.trim_end_matches('\n') {
        return Err("Ceiling result mismatch".to_owned());
    }
    Ok(elapsed.as_nanos() as u64)
}

fn execute(entry: EntryFunction, bytes: &[u8]) -> Result<String, String> {
    let configuration = parse(bytes).ok_or("Missing configuration")?;
    let mut pairs = Vec::new();
    for iteration in 0..configuration.iterations {
        let first = configuration.process.wrapping_add(iteration) & 1 == 0;
        let (native_time, ceiling_time) = if first {
            let native_time = native(entry, &configuration)?;
            (native_time, ceiling(&configuration)?)
        } else {
            let ceiling_time = ceiling(&configuration)?;
            (native(entry, &configuration)?, ceiling_time)
        };
        pairs.push(serde_json::json!({"iteration": iteration, "native_first": first, "native_nanoseconds": native_time, "ceiling_nanoseconds": ceiling_time, "vm_nanoseconds": 0}));
    }
    serde_json::to_string(&pairs).map_err(|error| error.to_string())
}

#[unsafe(no_mangle)]
pub extern "C" fn pseudolearn_native_pair(entry: EntryFunction, configuration: *const u8, count: usize) -> i32 {
    if configuration.is_null() || count == 0 {
        return 1;
    }
    let bytes = unsafe { std::slice::from_raw_parts(configuration, count) };
    match execute(entry, bytes) {
        Ok(document) => {
            println!("{document}");
            0
        }
        Err(error) => {
            eprintln!("{error}");
            1
        }
    }
}
